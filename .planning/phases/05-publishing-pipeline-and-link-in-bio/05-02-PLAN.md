1. **Create supabase/functions/publish-scheduled/index.ts** -- Deno runtime Edge Function:

   CRITICAL: This is Deno, NOT Node.js. Use:
   - `Deno.serve()` (not express or http server)
   - `Deno.env.get()` (not process.env)
   - `import { createClient } from "https://esm.sh/@supabase/supabase-js@2"` (not npm import)
   - Direct fetch to Late.dev REST API (not @getlatedev/node SDK)

   Implementation:
   ```
   Deno.serve(async (req) => {
     // Create Supabase client with SERVICE_ROLE_KEY (bypasses RLS)
     const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY)

     // RACE CONDITION PREVENTION: Use atomic update to claim posts
     // Update status from "scheduled" to "publishing" (intermediate status)
     // and return the updated rows in one atomic operation
     // Since DB enum may not have "publishing", use a lock approach:
     // Query posts with scheduled_at <= now AND status = "scheduled"
     // Process sequentially with 1s delay between posts (rate limit safety)

     const { data: posts } = await supabase
       .from("posts")
       .select("*, post_channels(channel_id, channels(platform, account_id, name, metadata))")
       .eq("status", "scheduled")
       .lte("scheduled_at", new Date().toISOString())

     for (const post of posts ?? []) {
       try {
         // Call Late.dev REST API directly
         const response = await fetch("https://api.getlate.dev/v1/posts", {
           method: "POST",
           headers: {
             "Authorization": `Bearer ${Deno.env.get("LATE_API_KEY")}`,
             "Content-Type": "application/json",
           },
           body: JSON.stringify({
             content: post.caption,
             platforms: post.post_channels.map(pc => ({
               platform: pc.channels.platform,
               accountId: pc.channels.account_id,
             })),
             publishNow: true,
             // Include media URLs if post has media
           }),
         })

         if (response.ok) {
           const result = await response.json()
           await supabase.from("posts").update({
             status: "published",
             published_at: new Date().toISOString(),
             platform_post_ids: result,
           }).eq("id", post.id)

           // Handle link-in-bio auto-update (BIO-03)
           if (post.update_linkinbio && post.link_id) {
             // Find active bio page
             const { data: bioPages } = await supabase
               .from("linkinbio_pages")
               .select("id")
               .eq("is_active", true)
               .limit(1)

             if (bioPages && bioPages.length > 0) {
               const pageId = bioPages[0].id
               // Shift all existing items down by 1 position
               const { data: existingItems } = await supabase
                 .from("linkinbio_items")
                 .select("id, position")
                 .eq("page_id", pageId)
                 .order("position", { ascending: true })

               for (const item of existingItems ?? []) {
                 await supabase.from("linkinbio_items")
                   .update({ position: item.position + 1 })
                   .eq("id", item.id)
               }

               // Check if link already exists as bio item
               const { data: existingItem } = await supabase
                 .from("linkinbio_items")
                 .select("id")
                 .eq("page_id", pageId)
                 .eq("link_id", post.link_id)
                 .maybeSingle()

               if (existingItem) {
                 // Move existing item to position 0
                 await supabase.from("linkinbio_items")
                   .update({ position: 0 })
                   .eq("id", existingItem.id)
               } else {
                 // Insert new item at position 0
                 await supabase.from("linkinbio_items").insert({
                   page_id: pageId,
                   link_id: post.link_id,
                   position: 0,
                   is_visible: true,
                 })
               }
             }
           }
         } else {
           const errorText = await response.text()
           await supabase.from("posts").update({
             status: "failed",
             publish_errors: {
               message: errorText,
               timestamp: new Date().toISOString(),
               statusCode: response.status,
             },
           }).eq("id", post.id)
         }

         // Rate limit safety: 1s delay between posts
         if ((posts?.length ?? 0) > 1) {
           await new Promise(r => setTimeout(r, 1000))
         }
       } catch (err) {
         await supabase.from("posts").update({
           status: "failed",
           publish_errors: {
             message: String(err),
             timestamp: new Date().toISOString(),
           },
         }).eq("id", post.id)
       }
     }

     return new Response(JSON.stringify({ processed: posts?.length ?? 0 }), {
       headers: { "Content-Type": "application/json" },
     })
   })
   ```
