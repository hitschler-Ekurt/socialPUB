   - `getWeekHours(date: Date): { day: Date; hours: Date[] }[]` -- 7 days x 24 hourly slots
   - `getDayHours(date: Date): Date[]` -- 24 hourly slots for a single day
   - `groupPostsByDay(posts: Post[]): Map<string, Post[]>` -- key is "yyyy-MM-dd", uses parseISO for timezone-safe day matching
   - `formatMonthYear(date: Date): string` -- e.g. "Maerz 2026" using date-fns de locale
   - `formatWeekRange(date: Date): string` -- e.g. "10. - 16. Maerz 2026"
   - `formatDayHeader(date: Date): string` -- e.g. "Donnerstag, 13. Maerz"
