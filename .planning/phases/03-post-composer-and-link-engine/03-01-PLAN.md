```sql
CREATE TABLE public.links (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  original_url TEXT NOT NULL,
  short_code TEXT UNIQUE NOT NULL,            -- auto-generated by trigger
  short_url TEXT GENERATED ALWAYS AS ('https://link.hitschies.de/' || short_code) STORED,
  utm_source TEXT, utm_medium TEXT DEFAULT 'social', utm_campaign TEXT, utm_content TEXT, utm_term TEXT,
  full_url TEXT,                               -- auto-computed by trigger
  title TEXT,
  campaign_id UUID REFERENCES public.campaigns(id),
  expires_at TIMESTAMPTZ, is_active BOOLEAN DEFAULT true,
  click_count INTEGER DEFAULT 0,
  created_by UUID, created_at TIMESTAMPTZ, updated_at TIMESTAMPTZ
);
-- Triggers: generate_short_code() on INSERT, build_full_url() on INSERT/UPDATE
```
